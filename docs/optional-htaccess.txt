# OPTIONAL server config for Hostinger (Apache / LiteSpeed).
#
# NOT ACTIVE, on purpose. The deployment confirmed working on Hostinger did
# not contain an .htaccess, so it was pulled out of the build to keep the
# deployed tree byte-identical to the known-good one. The site works fine
# without it -- everything below is polish (custom 404, caching,
# compression), never required.
#
# TO ENABLE, once the site is confirmed live and stable:
#   1. Copy this file to  public/.htaccess
#   2. Commit and push -- the export copies public/ to the web root, so it
#      lands as /.htaccess on the server.
#   3. Load the site straight away. If anything 403s or 500s, delete
#      public/.htaccess, push again, and the site recovers.
# If a directive turns out to be unsupported on the plan, add the blocks
# back one at a time (start with just ErrorDocument) to isolate it.
#
# Note: clean URLs need no rewrite rules. The site is exported with
# trailingSlash, so /apply is a real directory (apply/index.html) and the
# server's DirectoryIndex serves it natively.

# --- Custom error page -------------------------------------------------------
ErrorDocument 404 /404.html

# --- Compression -------------------------------------------------------------
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/css text/javascript
  AddOutputFilterByType DEFLATE application/javascript application/json
  AddOutputFilterByType DEFLATE image/svg+xml
</IfModule>

# --- Caching -----------------------------------------------------------------
<IfModule mod_headers.c>
  # Next.js fingerprints these filenames, so a long immutable cache is safe:
  # a new build produces new filenames rather than changing old ones.
  <FilesMatch "\.(js|css|woff2)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>

  # HTML must revalidate, or visitors keep seeing the previous deploy.
  <FilesMatch "\.html$">
    Header set Cache-Control "public, max-age=0, must-revalidate"
  </FilesMatch>

  # Images/PDFs change rarely but aren't fingerprinted - cache a day.
  <FilesMatch "\.(jpg|jpeg|png|svg|ico|webp|pdf|docx)$">
    Header set Cache-Control "public, max-age=86400"
  </FilesMatch>

  # Basic hardening.
  Header set X-Content-Type-Options "nosniff"
  Header set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>
